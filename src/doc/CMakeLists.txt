# The thread table of the hole feature (task T-097), compiled in so a build tree and an installed program find it
# alike (as src/ui compiles in the themes). Changing the JSON re-runs CMake, which rewrites the generated source.
set(RAY_THREAD_TABLE "${CMAKE_SOURCE_DIR}/resources/standards/threads.json")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${RAY_THREAD_TABLE}")
file(READ "${RAY_THREAD_TABLE}" RAY_THREAD_TABLE_HEX HEX)
string(LENGTH "${RAY_THREAD_TABLE_HEX}" RAY_THREAD_TABLE_HEX_LENGTH)
math(EXPR RAY_THREAD_TABLE_SIZE "${RAY_THREAD_TABLE_HEX_LENGTH} / 2")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," RAY_THREAD_TABLE_BYTES "${RAY_THREAD_TABLE_HEX}")
string(REGEX REPLACE "((0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],))" "\\1\n"
    RAY_THREAD_TABLE_BYTES "${RAY_THREAD_TABLE_BYTES}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/thread_table_data.cpp" CONTENT
"// Generated by src/doc/CMakeLists.txt from resources/standards/threads.json. Do not edit.
#include <cstddef>
namespace ray::doc::features::detail {
extern const unsigned char kThreadTableJson[];
extern const std::size_t kThreadTableJsonSize;
const unsigned char kThreadTableJson[] = {
@RAY_THREAD_TABLE_BYTES@0x00};
const std::size_t kThreadTableJsonSize = @RAY_THREAD_TABLE_SIZE@;
} // namespace ray::doc::features::detail
" @ONLY)
target_sources(ray_doc PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/thread_table_data.cpp")
target_link_libraries(ray_doc PRIVATE ray_core ray_kernel ray_naming ray_sketch ray_cam)
