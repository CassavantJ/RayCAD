# lib: Tool, holder, material, machine, template and rule libraries. Allowed dependencies: PLAN.md section 1.1.
add_library(ray_lib STATIC
    codec.cpp
    fileio.cpp
    json.cpp
    json_number.cpp
    lib_module.cpp
    library.cpp
    schema.cpp
    stack.cpp
    tool_query.cpp
)
ray_module(ray_lib DETERMINISTIC)
target_link_libraries(ray_lib PRIVATE ray_core)

# The JSON Schemas of the library files (task T-030), compiled in so a build tree and an installed program check
# files alike (as src/doc compiles in the thread table). Changing a schema re-runs CMake, which rewrites the source.
set(RAY_LIB_SCHEMAS tool holder material machine optemplate)
set(_entries "")
set(_arrays "")
foreach(_name IN LISTS RAY_LIB_SCHEMAS)
    set(_file "${CMAKE_SOURCE_DIR}/schemas/${_name}.schema.json")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_file}")
    file(READ "${_file}" _hex HEX)
    string(LENGTH "${_hex}" _hex_length)
    math(EXPR _size "${_hex_length} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
    # 16 bytes per line keeps the generated source readable and its lines short.
    string(REGEX REPLACE "((0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],))" "\\1\n"
        _bytes "${_bytes}")
    string(APPEND _arrays "const unsigned char kSchema_${_name}[] = {\n${_bytes}0x00};\n")
    string(APPEND _entries "    {\"${_name}\", kSchema_${_name}, ${_size}},\n")
endforeach()
list(LENGTH RAY_LIB_SCHEMAS _count)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/lib_schemas_data.cpp" CONTENT
"// Generated by src/lib/CMakeLists.txt from schemas/*.schema.json. Do not edit.
#include <cstddef>
namespace ray::lib::detail {
struct EmbeddedFile {
    const char* name;
    const unsigned char* data;
    std::size_t size;
};
extern const EmbeddedFile kEmbeddedSchemas[];
extern const std::size_t kEmbeddedSchemaCount;
namespace {
@_arrays@} // namespace
const EmbeddedFile kEmbeddedSchemas[] = {
@_entries@};
const std::size_t kEmbeddedSchemaCount = @_count@;
} // namespace ray::lib::detail
" @ONLY)
target_sources(ray_lib PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/lib_schemas_data.cpp")
