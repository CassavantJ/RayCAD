# post: Post engine, G-code output, setup sheet. Allowed dependencies: PLAN.md section 1.1.
add_library(ray_post STATIC
    post_module.cpp
    format.h format.cpp
    writer.h writer.cpp
    engine.h engine.cpp
    engine_host.h engine_host.cpp
    engine_out.cpp
    engine_program.h engine_program.cpp

# T-076: the setup sheet's print style (resources/setupsheet/setupsheet.css), compiled in as src/cam compiles its
# schemas and font, so a sheet never depends on a file on disk. Changing the file re-runs CMake, which rewrites the source.
set(_sheet_css "${CMAKE_SOURCE_DIR}/resources/setupsheet/setupsheet.css")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_sheet_css}")
file(READ "${_sheet_css}" _hex HEX)
string(LENGTH "${_hex}" _hex_length)
math(EXPR _size "${_hex_length} / 2")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
string(REGEX REPLACE "((0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],))" "\\1\n"
    _bytes "${_bytes}")
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/post_setupsheet_css.cpp" CONTENT
"// Generated by src/post/CMakeLists.txt from resources/setupsheet/setupsheet.css. Do not edit.
#include <cstddef>
namespace ray::post::detail {
extern const unsigned char kSetupSheetCss[];
extern const std::size_t kSetupSheetCssSize;
const unsigned char kSetupSheetCss[] = {
@_bytes@0x00};
const std::size_t kSetupSheetCssSize = @_size@;
} // namespace ray::post::detail
" @ONLY)
target_sources(ray_post PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/post_setupsheet_css.cpp")

# QuickJS-NG 0.18 from the overlay port ports/quickjs-ng: the plain qjs library (no quickjs-libc), with the
# interrupt poll patched to every 100 back-edges (THIRD_PARTY.md). Static; only src/post includes quickjs.h.
# (The post thread's pthread_create needs no library of its own: glibc 2.34 and later and macOS have it in libc,
# and qjs already links the platform's thread library.)
find_package(qjs CONFIG REQUIRED)
target_link_libraries(ray_post PRIVATE qjs)
