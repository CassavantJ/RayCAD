    feeds.cpp    # computeFeeds (task T-048)
    holes.cpp    # recognizeHoles, orderHolesShortestPath (task T-034)
    json.cpp     # the JSON reader of operation params (a copy of src/lib's; task T-035)
    json.h

# The parameter schemas of the operation types (schemas/ops/<type>.schema.json), compiled in so the type returns its
# schema (IOperationType::paramsSchemaJson) and reads its defaults from the same text (ops/op_params.h), as src/lib
# compiles in its schemas. Changing a schema re-runs CMake, which rewrites the source. A new type adds its name here.
set(RAY_CAM_OP_SCHEMAS drill)
set(_entries "")
set(_arrays "")
foreach(_name IN LISTS RAY_CAM_OP_SCHEMAS)
    set(_file "${CMAKE_SOURCE_DIR}/schemas/ops/${_name}.schema.json")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_file}")
    file(READ "${_file}" _hex HEX)
    string(LENGTH "${_hex}" _hex_length)
    math(EXPR _size "${_hex_length} / 2")
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," _bytes "${_hex}")
    string(REGEX REPLACE "((0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],)(0x[0-9a-f][0-9a-f],))" "\\1\n"
        _bytes "${_bytes}")
    string(APPEND _arrays "const unsigned char kOpSchema_${_name}[] = {\n${_bytes}0x00};\n")
    string(APPEND _entries "    {\"${_name}\", kOpSchema_${_name}, ${_size}},\n")
endforeach()
list(LENGTH RAY_CAM_OP_SCHEMAS _count)
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/generated/cam_op_schemas_data.cpp" CONTENT
"// Generated by src/cam/CMakeLists.txt from schemas/ops/*.schema.json. Do not edit.
#include <cstddef>
namespace ray::cam::ops::detail {
struct EmbeddedOpSchema {
    const char* name;
    const unsigned char* data;
    std::size_t size;
};
extern const EmbeddedOpSchema kEmbeddedOpSchemas[];
extern const std::size_t kEmbeddedOpSchemaCount;
namespace {
@_arrays@} // namespace
const EmbeddedOpSchema kEmbeddedOpSchemas[] = {
@_entries@};
const std::size_t kEmbeddedOpSchemaCount = @_count@;
} // namespace ray::cam::ops::detail
" @ONLY)
target_sources(ray_cam PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/cam_op_schemas_data.cpp")
