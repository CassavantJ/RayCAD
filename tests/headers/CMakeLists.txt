# Every public header must compile on its own, in its own translation unit, with nothing included before it.
file(GLOB _ray_public_headers CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/include/ray/*.h")
list(SORT _ray_public_headers)
set(_ray_header_tus)
set(_decls "")
set(_items "")
foreach(_header IN LISTS _ray_public_headers)
    get_filename_component(_name "${_header}" NAME_WE)
    get_filename_component(_file "${_header}" NAME)
    set(_tu "${CMAKE_CURRENT_BINARY_DIR}/header_${_name}.cpp")
    # Included twice: the second include checks the include guard.
    file(CONFIGURE OUTPUT "${_tu}" CONTENT
"// Generated by tests/headers/CMakeLists.txt. Compiles <ray/${_file}> standalone.
#include <ray/${_file}>
#include <ray/${_file}>

extern const char* const ray_header_tu_${_name};
const char* const ray_header_tu_${_name} = \"${_file}\";
" @ONLY)
    list(APPEND _ray_header_tus "${_tu}")
    string(APPEND _decls "extern const char* const ray_header_tu_${_name};\n")
    string(APPEND _items "        ray_header_tu_${_name},\n")
endforeach()

file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/header_list.cpp" CONTENT
"// Generated by tests/headers/CMakeLists.txt.
#include <cstdio>

${_decls}
int main()
{
    const char* const headers[] = {
${_items}    };
    int count = 0;
    for (const char* header : headers) {
        std::printf(\"compiled standalone: ray/%s\\n\", header);
        ++count;
    }
    std::printf(\"%d public headers\\n\", count);
    return count > 0 ? 0 : 1;
}
" @ONLY)

# Links ray_headers only: no module library, so a header cannot lean on another module's build settings.
add_executable(ray_header_tests ${_ray_header_tus} "${CMAKE_CURRENT_BINARY_DIR}/header_list.cpp")
target_link_libraries(ray_header_tests PRIVATE ray_headers)
ray_add_test(ray_header_tests)
