cmake_minimum_required(VERSION 3.24)
project(s10_sqlite_smb C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SQLITE_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/sqlite-amalgamation-3540100)
add_library(sqlite3 STATIC ${SQLITE_DIR}/sqlite3.c)
target_include_directories(sqlite3 PUBLIC ${SQLITE_DIR})
add_executable(s10 main.cpp)
target_link_libraries(s10 PRIVATE sqlite3)
target_compile_definitions(s10 PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX)
